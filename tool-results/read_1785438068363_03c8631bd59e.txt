   143→function getTimeAgo(dateStr: string): string {
   144→  const diff = Date.now() - new Date(dateStr).getTime();
   145→  const mins = Math.floor(diff / 60000);
   146→  if (mins < 1) return "à l'instant";
   147→  if (mins < 60) return `il y a ${mins}min`;
   148→  const hours = Math.floor(mins / 60);
   149→  if (hours < 24) return `il y a ${hours}h`;
   150→  const days = Math.floor(hours / 24);
   151→  return `il y a ${days}j`;
   152→}
