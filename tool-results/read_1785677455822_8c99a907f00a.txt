   167→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   168→  const ref = useRef(null);
   169→  const isInView = useInView(ref, { once: true, margin: '-80px' });
   170→  return (
   171→    <motion.section
   172→      ref={ref}
   173→      id={id}
   174→      initial="hidden"
   175→      animate={isInView ? 'visible' : 'hidden'}
   176→      variants={stagger}
   177→      className={className}
   178→    >
   179→      {children}
   180→    </motion.section>
   181→  );
   182→}
