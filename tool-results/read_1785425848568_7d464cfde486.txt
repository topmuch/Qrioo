   100→function getCategoryIcon(category?: string | null): string {
   101→  if (!category) return '📦';
   102→  const key = category.toLowerCase().trim();
   103→  for (const [k, v] of Object.entries(CATEGORY_ICONS)) {
   104→    if (key.includes(k)) return v;
   105→  }
   106→  return '📦';

   109→// ─── Hook minimal : détection du pays via IP ───
   110→function useDetectedCountry(): { countryCode: string; isLoading: boolean } {
   111→  const [countryCode, setCountryCode] = useState('FR');
   112→  const [isLoading, setIsLoading] = useState(true);
   113→
   114→  useEffect(() => {
   115→    let cancelled = false;
   116→    (async () => {
   117→      try {
   118→        const res = await fetch('/api/detect-country', { cache: 'no-store' });
   119→        if (res.ok) {
   120→          const data = await res.json();
   121→          if (data?.countryCode && !cancelled) {
   122→            setCountryCode(String(data.countryCode).toUpperCase());
   123→          }
   124→        }
   125→      } catch {
   126→        // Silent fallback to FR
   127→      } finally {
   128→        if (!cancelled) setIsLoading(false);
   129→      }
   130→    })();
   131→    return () => { cancelled = true; };
   132→  }, []);
   133→
   134→  return { countryCode, isLoading };
   135→}

   137→function isLocalPhoneValid(localDigits: string): boolean {
   138→  const n = localDigits.replace(/\D/g, '');
   139→  return n.length >= 6 && n.length <= 15;

   142→function getMonthlyFoundCount(): number {
   143→  const month = new Date().getMonth();
   144→  return 127 + ((month * 13) % 40);
   145→}
