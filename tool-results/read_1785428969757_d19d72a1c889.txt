    86→function formatDate(iso: string | null | undefined): string {
    87→  if (!iso) return '—';
    88→  try {
    89→    return new Date(iso).toLocaleString('fr-FR', {
    90→      dateStyle: 'long',
    91→      timeStyle: 'short',
    92→    });
    93→  } catch {
    94→    return '—';
    95→  }
